def self_attention(query, key, value):
    d_k = query.size(-1)
    # Compute attention scores
    scores = torch.matmul(query, key.transpose(-2, -1)) / math.sqrt(d_k)
    # Apply softmax
    p_attn = F.softmax(scores, dim=-1)
    # Multiply by values
    return torch.matmul(p_attn, value)
